```python
import requests

knowledge_bases = [
    {
        "name": "企业架构知识库",
        "description": "企业架构总览和跨架构域文档",

    },

        "status": "active"
    },
    {
        "name": "应用架构知识库",
        "description": "应用系统、应用服务、API接口文档",
        "status": "active"
    },
    {
        "name": "数据架构知识库",
        "description": "数据模型、数据实体、数据流文档",
        "status": "active"
    },
    {
        "name": "技术架构知识库",
        "description": "技术栈、基础设施、网络架构文档",
        "status": "active"
    }
]

def create_knowledge_bases():
    """创建企业架构知识库"""
    created_kbs = {}
    for kb in knowledge_bases:
        response = requests.post(f"{BASE_URL}/knowledge-bases", json=kb)
        if response.status_code in [200, 201]:
            result = response.json()
            created_kbs[kb["name"]] = result.get("id")
            print(f"✅ 创建知识库: {kb['name']} (ID: {result.get('id')})")
        else:
            print(f"❌ 创建知识库失败: {kb['name']} - {response.text}")
    return created_kbs

if __name__ == "__main__":
    create_knowledge_bases()
```

#### 导入脚本

```python
# import_ea_documents.py
import requests
import os
import json
from pathlib import Path

BASE_URL = "http://localhost:8004/api"
DOCS_DIR = "enterprise_architecture_docs"

# 知识库ID映射（从创建脚本获取）
KB_IDS = {
    "企业架构知识库": "kb-id-1",
    "业务架构知识库": "kb-id-2",
    "应用架构知识库": "kb-id-3",
    "数据架构知识库": "kb-id-4",
    "技术架构知识库": "kb-id-5"
}

def import_documents():
    """导入企业架构文档"""
    for kb_name, kb_id in KB_IDS.items():
        kb_dir = Path(DOCS_DIR) / kb_name.replace("知识库", "").lower()
        if not kb_dir.exists():
            continue
        
        print(f"\n导入 {kb_name} 的文档...")
        for doc_file in kb_dir.rglob("*.md"):
            with open(doc_file, "r", encoding="utf-8") as f:
                content = f.read()
            
            doc_data = {
                "title": doc_file.stem,
                "content": content,
                "knowledge_base_id": kb_id,
                "category": doc_file.parent.name,
                "tags": [kb_name, doc_file.parent.name],
                "process_async": True
            }
            
            response = requests.post(
                f"{BASE_URL}/documents/create",
                json=doc_data
            )
            
            if response.status_code in [200, 201]:
                print(f"  ✅ {doc_file.name}")
            else:
                print(f"  ❌ {doc_file.name} - {response.text}")

if __name__ == "__main__":
    import_documents()
```

#### 架构关系表

```sql
-- 架构关系表
CREATE TABLE architecture_relationships (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    source_type VARCHAR(100) NOT NULL,  -- 源类型: business_process, application_system, etc.
    source_id UUID NOT NULL,
    target_type VARCHAR(100) NOT NULL,   -- 目标类型
    target_id UUID NOT NULL,
    relationship_type VARCHAR(100) NOT NULL,  -- 关系类型: implements, uses, depends_on, etc.
    description TEXT,
    metadata JSONB,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 创建索引
CREATE INDEX idx_arch_rels_source ON architecture_relationships(source_type, source_id);
CREATE INDEX idx_arch_rels_target ON architecture_relationships(target_type, target_id);
CREATE INDEX idx_arch_rels_type ON architecture_relationships(relationship_type);
```

### 2.2 API实现

#### EnterpriseArchitectureService

```python
# services/enterprise_architecture_service.py
from typing import Dict, List, Optional, Any
from sqlalchemy.orm import Session
from sqlalchemy import and_, or_

class EnterpriseArchitectureService:
    def __init__(self, db: Session):
        self.db = db
    
    async def get_overview(self) -> Dict[str, Any]:
        """获取企业架构总览"""
        return {
            "business_architecture": {
                "processes_count": self._count_business_processes(),
                "capabilities_count": self._count_business_capabilities(),
                "services_count": self._count_business_services()
            },
            "application_architecture": {
                "systems_count": self._count_application_systems(),
                "services_count": self._count_application_services()
            },
            "data_architecture": {
                "entities_count": self._count_data_entities(),
                "models_count": self._count_data_models()
            },
            "technology_architecture": {
                "components_count": self._count_technology_components(),
                "stacks_count": self._count_technology_stacks()
            }
        }
    
    async def get_business_architecture(self) -> Dict[str, Any]:
        """获取业务架构"""
        return {
            "processes": self._get_business_processes(),
            "capabilities": self._get_business_capabilities(),
            "services": self._get_business_services()
        }
    
    async def get_architecture_graph(self) -> Dict[str, Any]:
        """获取架构关系图"""
        nodes = []
        links = []
        
        # 添加业务架构节点
        processes = self._get_business_processes()
        for process in processes:
            nodes.append({
                "id": str(process["id"]),
                "name": process["name"],
                "type": "business_process",
                "group": "business"
            })
        
        # 添加应用架构节点
        systems = self._get_application_systems()
        for system in systems:
            nodes.append({
                "id": str(system["id"]),
                "name": system["name"],
                "type": "application_system",
                "group": "application"
            })
        
        # 添加关系边
        relationships = self._get_architecture_relationships()
        for rel in relationships:
            links.append({
                "source": str(rel["source_id"]),
                "target": str(rel["target_id"]),
                "type": rel["relationship_type"]
            })
        
        return {"nodes": nodes, "links": links}
    
    async def analyze_impact(self, entity_id: str, entity_type: str) -> Dict[str, Any]:
        """分析架构影响"""
        # 查找依赖
        dependencies = self._get_dependencies(entity_id, entity_type)
        # 查找影响
        impacts = self._get_impacts(entity_id, entity_type)
        
        return {
            "dependencies": dependencies,
            "impacts": impacts,
            "risk_level": self._calculate_risk_level(dependencies, impacts)
        }
```
