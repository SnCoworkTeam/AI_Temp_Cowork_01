## 一、需求分析

### 1.1 功能需求

1. **项目计划模板系统**
   - 模板包含项目阶段（从基础数据获取）
   - 模板可以创建、编辑、删除、查看
   - 模板结构以JSON格式存储

2. **基于模板创建计划**
   - 创建计划时可以选择模板
   - 选择模板后，自动生成包含阶段的计划结构
   - 计划继承模板的阶段结构

3. **树形结构展示**
   - 项目计划以树形结构展示
   - 树节点分为两类：项目阶段（Phase）和项目任务（Task）
   - 阶段作为父节点，任务作为子节点

4. **节点操作权限**
   - **项目阶段**：不可增删改（只读），数据来源于基础数据
   - **项目任务**：可以增删改查

### 1.2 业务规则

- 项目阶段数据来源于基础数据分类（`category_type = 'project_phase'`）
- 阶段顺序由基础数据的`sort_order`决定
- 任务必须归属于某个阶段
- 模板中的阶段是固定的，不能修改
- 计划创建后，可以添加、修改、删除任务，但不能修改阶段

```python
# project-management/src/routes/project_templates.py

from fastapi import APIRouter, HTTPException, Depends
from typing import List, Optional
from pydantic import BaseModel
from sqlalchemy.orm import Session

router = APIRouter()

class ProjectTemplateResponse(BaseModel):
    id: str
    template_code: str
    name: str
    description: Optional[str]
    category_id: Optional[str]
    template_structure: dict
    is_active: bool
    usage_count: int

class ProjectTemplateCreate(BaseModel):
    template_code: str
    name: str
    description: Optional[str] = None
    category_id: Optional[str] = None
    phase_category_ids: List[str]  # 阶段分类ID列表

@router.get("/project-templates", response_model=List[ProjectTemplateResponse])
async def list_templates(
    is_active: Optional[bool] = None,
    db: Session = Depends(get_db),
    current_user: dict = Depends(require_auth)
):
    """获取模板列表"""
    query = db.query(ProjectTemplate)
    if is_active is not None:
        query = query.filter(ProjectTemplate.is_active == is_active)
    templates = query.order_by(ProjectTemplate.created_at.desc()).all()
    return templates

@router.post("/project-templates", response_model=ProjectTemplateResponse, status_code=201)
async def create_template(
    template_data: ProjectTemplateCreate,
    db: Session = Depends(get_db),
    current_user: dict = Depends(require_auth)
):
    """创建模板"""
    # 验证阶段分类ID
    phases = []
    for idx, category_id in enumerate(template_data.phase_category_ids):
        category = db.query(BasicDataCategory).filter(
            BasicDataCategory.id == UUID(category_id),
            BasicDataCategory.category_type == 'project_phase'
        ).first()
        if not category:
            raise HTTPException(status_code=400, detail=f"无效的阶段分类ID: {category_id}")

        phases.append({
            "category_id": str(category.id),
            "category_code": category.code,
            "category_name": category.name,
            "sequence": idx + 1,
            "tasks": []
        })

    template = ProjectTemplate(
        template_code=template_data.template_code,
        name=template_data.name,
        description=template_data.description,
        category_id=UUID(template_data.category_id) if template_data.category_id else None,
        template_structure={"phases": phases},
        is_active=True
    )

    db.add(template)
    db.commit()
    db.refresh(template)

    return template
```

```python
                        category_id=category_id,
                        name=category.name,
                        sequence=phase_data.get("sequence", 0),
                        description=f"从模板'{template.name}'自动创建"
                    )
                    db.add(phase)

    db.commit()
    db.refresh(plan)
```

#### 5.1.3 树形结构接口

```python
# project-management/src/routes/project_plans.py

class PlanTreeNode(BaseModel):
    id: str
    type: str  # 'phase' or 'task'
    name: str
    category_id: Optional[str] = None
    category_code: Optional[str] = None
    sequence: Optional[int] = None
    is_readonly: bool = False
    phase_id: Optional[str] = None
    status: Optional[str] = None
    progress_percent: Optional[float] = None
    children: List['PlanTreeNode'] = []

class PlanTreeResponse(BaseModel):
    plan_id: str
    plan_name: str
    tree: List[PlanTreeNode]

@router.get("/projects/{project_id}/plans/{plan_id}/tree", response_model=PlanTreeResponse)
async def get_plan_tree(
    project_id: str,
    plan_id: str,
    db: Session = Depends(get_db),
    current_user: dict = Depends(require_auth)
):
    """获取计划树形结构"""
    # ... 权限检查 ...

    plan = db.query(ProjectPlan).filter(ProjectPlan.id == UUID(plan_id)).first()
    if not plan:
        raise HTTPException(status_code=404, detail="计划不存在")

    # 获取项目的所有阶段（按sequence排序）
    phases = db.query(ProjectPhase).filter(
        ProjectPhase.project_id == UUID(project_id)
    ).order_by(ProjectPhase.sequence.asc()).all()

    # 获取计划的所有任务
    tasks = db.query(ProjectPlanTask).filter(
        ProjectPlanTask.plan_id == UUID(plan_id)
    ).all()

    # 构建树形结构
    tree = []
    for phase in phases:
        phase_node = PlanTreeNode(
            id=str(phase.id),
            type="phase",
            name=phase.name,
            category_id=str(phase.category_id),
            category_code=phase.category.code if phase.category else None,
            sequence=phase.sequence,
            is_readonly=True,
            children=[]
        )

        # 添加该阶段下的任务
        phase_tasks = [t for t in tasks if t.phase_id == phase.id]
        for task in phase_tasks:
            task_node = PlanTreeNode(
                id=str(task.id),
                type="task",
                name=task.name,
                phase_id=str(task.phase_id),
                status=task.status,
                progress_percent=task.progress_percent,
                children=[]
            )
            phase_node.children.append(task_node)

        tree.append(phase_node)

    return PlanTreeResponse(
        plan_id=str(plan.id),
        plan_name=plan.name,
        tree=tree
    )
```

```typescript
        `${apiUrl}/api/v1/projects/${projectId}/plans/${planId}/tree`,
        {
          headers: {
            'Authorization': `Bearer ${token}`,
          },
        }
      )

      if (response.ok) {
        const data = await response.json()
        setTree(data.tree || [])
        // 默认展开所有节点
        const allNodeIds = getAllNodeIds(data.tree)
        setExpandedNodes(new Set(allNodeIds))
      }
    } catch (error) {
      console.error('获取树形结构失败:', error)
    } finally {
      setLoading(false)
    }
  }

  const getAllNodeIds = (nodes: PlanTreeNode[]): string[] => {
    const ids: string[] = []
    nodes.forEach(node => {
      ids.push(node.id)
      if (node.children && node.children.length > 0) {
        ids.push(...getAllNodeIds(node.children))
      }
    })
    return ids
  }

  const toggleNode = (nodeId: string) => {
    const newExpanded = new Set(expandedNodes)
    if (newExpanded.has(nodeId)) {
      newExpanded.delete(nodeId)
    } else {
      newExpanded.add(nodeId)
    }
    setExpandedNodes(newExpanded)
  }

  const renderNode = (node: PlanTreeNode, level: number = 0) => {
    const isExpanded = expandedNodes.has(node.id)
    const hasChildren = node.children && node.children.length > 0
    const isPhase = node.type === 'phase'
    const isReadonly = node.is_readonly || isPhase

    return (
      <div key={node.id} className="select-none">
        <div
          className={`flex items-center gap-2 py-2 px-3 hover:bg-gray-50 rounded-lg ${
            isPhase ? 'bg-blue-50' : ''
          }`}
          style={{ paddingLeft: `${level * 24 + 12}px` }}
        >
          {/* 展开/折叠图标 */}
          {hasChildren ? (
            <button
              onClick={() => toggleNode(node.id)}
              className="p-1 hover:bg-gray-200 rounded"
            >
              {isExpanded ? (
                <ChevronDown className="w-4 h-4" />
              ) : (
                <ChevronRight className="w-4 h-4" />
              )}
            </button>
          ) : (
            <div className="w-6" />
          )}

          {/* 节点图标 */}
          {isPhase ? (
            <Folder className="w-5 h-5 text-blue-600" />
          ) : (
            <File className="w-5 h-5 text-gray-600" />
          )}

          {/* 节点名称 */}
          <span className={`flex-1 ${isReadonly ? 'text-gray-600' : 'text-gray-900'}`}>
            {node.name}
          </span>

          {/* 进度显示（任务节点） */}
          {!isPhase && node.progress_percent !== undefined && (
            <span className="text-sm text-gray-500">
              {node.progress_percent}%
            </span>
          )}

          {/* 操作按钮（仅任务节点） */}
          {!isReadonly && onNodeAction && (
            <div className="flex items-center gap-1">
              <button
                onClick={() => onNodeAction('add', node)}
                className="p-1 hover:bg-blue-100 rounded text-blue-600"
                title="添加子任务"
              >
                <Plus className="w-4 h-4" />
              </button>
              <button
                onClick={() => onNodeAction('edit', node)}
                className="p-1 hover:bg-green-100 rounded text-green-600"
                title="编辑"
              >
                <Edit className="w-4 h-4" />
              </button>
              <button
                onClick={() => onNodeAction('delete', node)}
                className="p-1 hover:bg-red-100 rounded text-red-600"
                title="删除"
              >
                <Trash2 className="w-4 h-4" />
              </button>
            </div>
          )}
        </div>

        {/* 子节点 */}
        {hasChildren && isExpanded && (
          <div>
            {node.children.map(child => renderNode(child, level + 1))}
          </div>
        )}
      </div>
    )
  }

  if (loading) {
    return <div className="text-center py-8 text-gray-500">加载中...</div>
  }

  return (
    <div className="bg-white rounded-xl shadow-sm border border-gray-200 p-6">
      <div className="space-y-1">
        {tree.map(node => renderNode(node))}
      </div>
    </div>
  )
}
```
