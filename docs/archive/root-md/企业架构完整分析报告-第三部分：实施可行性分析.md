#### 5.1.1 数据同步机制

```python
# metadata-service/src/services/data_sync_service.py

class DataSyncService:
    """数据同步服务"""
    
    async def sync_to_neo4j(self, entity_type: str, entity_id: str):
        """同步PostgreSQL数据到Neo4j"""
        # 1. 从PostgreSQL获取数据
        entity = await self.get_entity(entity_type, entity_id)
        
        # 2. 创建或更新Neo4j节点
        if entity.neo4j_node_id:
            await self.update_neo4j_node(entity)
        else:
            neo4j_id = await self.create_neo4j_node(entity)
            entity.neo4j_node_id = neo4j_id
            self.db.commit()
    
    async def check_data_consistency(self):
        """检查数据一致性"""
        inconsistencies = []
        
        # 检查PostgreSQL中有但Neo4j中没有的节点
        processes = self.db.query(BusinessProcess).filter(
            BusinessProcess.neo4j_node_id.isnot(None)
        ).all()
        
        for process in processes:
            exists = await self.check_neo4j_node_exists(process.neo4j_node_id)
            if not exists:
                inconsistencies.append({
                    "type": "BusinessProcess",
                    "id": str(process.id),
                    "issue": "Neo4j节点不存在"
                })
        
        return inconsistencies
```

```python
class Neo4jPerformanceOptimizer:

        """创建必要的索引"""
        indexes = [
            "CREATE INDEX IF NOT EXISTS FOR (p:BusinessProcess) ON (p.uuid)",

            "CREATE INDEX IF NOT EXISTS FOR (a:AIService) ON (a.uuid)",
            "CREATE INDEX IF NOT EXISTS FOR (o:OrganizationUnit) ON (o.uuid)",  # ⭐
            "CREATE INDEX IF NOT EXISTS FOR (p:BusinessProcess) ON (p.industry)",
            "CREATE INDEX IF NOT EXISTS FOR (p:BusinessProcess) ON (p.criticality)"
        ]
        
        for index_query in indexes:
            await self.neo4j.execute_write(index_query)
    
    async def batch_create_nodes(self, nodes: List[Dict]):
        """批量创建节点"""
        query = """
        UNWIND $nodes AS node
        CREATE (n:BusinessProcess)
        SET n = node
        RETURN n.uuid as uuid
        """
        
        await self.neo4j.execute_write(query, {"nodes": nodes})
```

```python
# metadata-service/src/services/data_quality_service.py

class DataQualityService:
    """数据质量服务"""
    
    async def validate_business_process(self, process: BusinessProcess) -> Dict:
        """验证业务流程数据质量"""
        quality_score = 100
        issues = []
        
        # 检查必填字段
        if not process.name:
            quality_score -= 20
            issues.append("流程名称缺失")
        
        # 检查痛点数据
        if not process.meta_data.get("pain_points"):
            quality_score -= 15
            issues.append("业务痛点未记录")
        
        # 检查关键性
        if not process.meta_data.get("criticality"):
            quality_score -= 10
            issues.append("关键性未标识")
        
        # 检查系统关联
        if not process.meta_data.get("sap_modules"):
            quality_score -= 10
            issues.append("SAP模块未关联")
        
        # 检查组织关联 ⭐
        if not process.organization_id:
            quality_score -= 10
            issues.append("组织关联缺失")
        
        return {
            "quality_score": quality_score,
            "issues": issues,
            "level": "high" if quality_score >= 80 else "medium" if quality_score >= 60 else "low"
        }
```
