# agent-service/src/core/conversation_gateway/gateway_core.py

class ConversationIntelligenceGateway:
    """对话智能网关核心"""

    def __init__(
        self,
        session_manager: SessionManager,
        intent_analyzer: IntentAnalyzer,
        format_converter: FormatConverter,
        knowledge_injector: KnowledgeInjector
    ):
        self.session_manager = session_manager
        self.intent_analyzer = intent_analyzer
        self.format_converter = format_converter
        self.knowledge_injector = knowledge_injector

    async def process_message(
        self,
        conversation_id: str,
        message: UserMessage
    ) -> AssistantMessage:
        """处理用户消息"""

        # 1. 获取/创建会话
        session = await self.session_manager.get_or_create(conversation_id)

        # 2. 意图识别
        intent = await self.intent_analyzer.analyze(message.content, session.context)

        # 3. 路由决策
        target_service = self._route_to_service(intent)

        # 4. 调用后端服务
        response = await self._call_service(target_service, message, intent, session)

        # 5. 知识注入
        enriched_response = await self.knowledge_injector.enrich(response, intent)

        # 6. 格式转换
        formatted_response = await self.format_converter.convert(enriched_response)

        # 7. 更新会话
        await self.session_manager.update(session, message, formatted_response)

        return formatted_response
```

# ea-core/src/services/business_semantic_engine.py

class BusinessSemanticEngine:
    """业务语义理解引擎"""

    def __init__(
        self,
        llm_client: LLMClient,
        neo4j_client: Neo4jClient,
        semantic_vectorizer: SemanticVectorizer
    ):
        self.llm = llm_client
        self.neo4j = neo4j_client
        self.vectorizer = semantic_vectorizer

    async def understand(self, user_query: str, context: Dict) -> BusinessIntent:
        """理解业务意图"""

        # 1. LLM初步理解
        llm_analysis = await self.llm.analyze(user_query, prompt="""
        分析以下用户查询，提取业务意图：

        查询：{query}

        输出JSON：
        {{
            "business_domain": "采购管理|销售管理|财务管理|...",
            "business_capability": "订单创建|报表查询|审批处理|...",
            "business_process": "具体流程名称",
            "entities": [实体列表],
            "actions": [动作列表]
        }}
        """)

        # 2. 从Neo4j检索相关业务架构
        business_arch = await self._retrieve_business_architecture(llm_analysis)

        # 3. 语义向量匹配
        similar_cases = await self._find_similar_cases(user_query)

        # 4. 融合理解结果
        business_intent = self._fuse_understanding(
            llm_analysis,
            business_arch,
            similar_cases,
            context
        )

        return business_intent

    async def _retrieve_business_architecture(
        self,
        llm_analysis: Dict
    ) -> Dict:
        """从Neo4j检索业务架构"""

        # Cypher查询
        query = """
        MATCH (domain:BusinessDomain {name: $domain_name})
        -[:HAS_CAPABILITY]->(cap:BusinessCapability {name: $capability_name})
        -[:IMPLEMENTS_PROCESS]->(process:BusinessProcess)
        -[:REQUIRES_DATA]->(data:DataAsset)
        RETURN domain, cap, process, data
        """

        result = await self.neo4j.run(query, {
            "domain_name": llm_analysis["business_domain"],
            "capability_name": llm_analysis["business_capability"]
        })

        return self._parse_business_arch(result)
```

```python
# agent-service/src/core/agent_executor.py

class ArchitectureAwareAgentExecutor:
    """架构感知的智能体执行器"""

    async def execute(
        self,
        agent_id: str,
        task: Task,
        ea_context: EAContext = None  # 新增EA上下文
    ) -> ExecutionResult:
        """执行智能体任务"""

        # 1. 获取智能体
        agent = await self.agent_registry.get(agent_id)

        # 2. 注入EA上下文（如果有）
        if ea_context:
            task = self._enrich_task_with_ea_context(task, ea_context)

        # 3. 执行任务
        result = await agent.run(task)

        # 4. 记录执行信息到EA（如果需要）
        if ea_context:
            await self._record_to_ea(ea_context, task, result)

        return result
```

### 7.2 关键技术决策

| 决策点 | 选项 | 推荐 | 理由 |
|-------|------|------|------|
| 图数据库 | Neo4j vs ArangoDB | Neo4j | 生态成熟，社区活跃 |
| 实时通信 | WebSocket vs SSE | WebSocket | 双向通信，功能更强 |
| 状态管理 | Redux vs Zustand | Zustand | 轻量级，易于使用 |
| 图谱查询 | Cypher vs Gremlin | Cypher | Neo4j原生，性能好 |
| 链路追踪 | OpenTelemetry vs Jaeger | OpenTelemetry | 标准协议，兼容性好 |
| EA引擎 | 规则引擎 vs LLM | 规则引擎 + LLM | MVP用规则，后续增强 |
