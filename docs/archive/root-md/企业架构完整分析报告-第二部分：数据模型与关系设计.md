```python
class TechnologySystem(BaseModel, TimestampMixin):

class AIService(BaseModel, TimestampMixin):
    """AI服务模型（属于AI平台技术系统）"""
    __tablename__ = "ai_services"
    
    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    system_id = Column(UUID(as_uuid=True), ForeignKey("technology_systems.id"), nullable=True, comment="所属技术系统ID（AI平台）")

    service_type = Column(String(100), nullable=True)
```

```python
# metadata-service/src/services/enterprise_architecture_graph_service.py

class EnterpriseArchitectureGraphService:
    """企业架构图服务"""
    
    async def create_business_process_node(
        self, process: BusinessProcess
    ) -> str:
        """创建业务流程节点"""
        node_id = await self.neo4j.create_node(
            labels=["BusinessProcess"],
            properties={
                "uuid": str(process.id),
                "name": process.name,
                "description": process.description,
                "criticality": process.meta_data.get("criticality"),
                "pain_points": process.meta_data.get("pain_points", []),
                "sap_modules": process.meta_data.get("sap_modules", []),
                "industry": process.meta_data.get("industry")
            }
        )
        
        process.neo4j_node_id = node_id
        self.db.commit()
        
        return node_id
```

### 3.2 关系创建服务

```python
async def link_business_to_system(
    self, process: BusinessProcess, system: ApplicationSystem,
    support_type: str = "主要支撑", coverage: float = 80.0
):
    """关联业务流程到应用系统"""
    if process.neo4j_node_id and system.neo4j_node_id:
        await self.neo4j.create_relationship(
            source_id=process.neo4j_node_id,
            target_id=system.neo4j_node_id,
            rel_type="SUPPORTED_BY",
            properties={
                "support_type": support_type,
                "coverage": coverage,
                "critical": True,
                "created_at": datetime.utcnow().isoformat()
            }
        )
```

```python
async def link_application_to_technology_system(
    self, app: ApplicationSystem, tech_system: TechnologySystem,
```

```python
async def create_organization_hierarchy(
    self, org_unit: OrganizationUnit
):
    """创建组织层级关系"""
    if org_unit.parent_id:
        parent = self.db.query(OrganizationUnit).filter(
            OrganizationUnit.id == org_unit.parent_id
        ).first()
        
        if parent and parent.neo4j_node_id and org_unit.neo4j_node_id:
            await self.neo4j.create_relationship(
                source_id=parent.neo4j_node_id,
                target_id=org_unit.neo4j_node_id,
                rel_type="PARENT_OF",
                properties={
                    "created_at": datetime.utcnow().isoformat()
                }
            )
```
