class AIPlatform(BaseModel, TimestampMixin):
    """AI平台模型"""
    __tablename__ = "ai_platforms"
    
    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    name = Column(String(255), nullable=False, index=True, comment="平台名称")
    platform_type = Column(String(100), nullable=True, comment="平台类型")
    version = Column(String(50), nullable=True, comment="版本")
    vendor = Column(String(255), nullable=True, comment="供应商")
    description = Column(Text, nullable=True, comment="描述")
    status = Column(String(50), nullable=True, default="active", comment="状态")
    neo4j_node_id = Column(String(255), nullable=True, comment="Neo4j节点ID")
    meta_data = Column(JSONB, nullable=True, default=dict, comment="扩展元数据")
    
    # 关系
    services = relationship("AIService", back_populates="platform", cascade="all, delete-orphan")


class AIService(BaseModel, TimestampMixin):

    __tablename__ = "ai_services"
    
    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    platform_id = Column(UUID(as_uuid=True), ForeignKey("ai_platforms.id"), nullable=True, index=True)

    meta_data = Column(JSONB, nullable=True, default=dict, comment="扩展元数据")
    
    # 关系

class EnterpriseArchitectureGraphService:

        self.neo4j = neo4j_client
    
    async def create_business_capability_node(
        self, capability: BusinessCapability
    ) -> str:
        """创建业务能力节点"""
        node_id = await self.neo4j.create_node(
            labels=["BusinessCapability"],
            properties={
                "uuid": str(capability.id),
                "name": capability.name,
                "description": capability.description,
                "level": capability.level,
                "industry": capability.meta_data.get("industry", "general")
            }
        )
        
        # 更新PostgreSQL记录
        capability.neo4j_node_id = node_id
        self.db.commit()
        
        return node_id
    
    async def create_business_process_node(
        self, process: BusinessProcess
    ) -> str:
        """创建业务流程节点"""

                "uuid": str(process.id),

async def create_business_relationships(self):
    """创建业务层关系"""
    # 1. 能力包含流程
    query = """
    MATCH (cap:BusinessCapability {uuid: $cap_id})
    MATCH (proc:BusinessProcess {uuid: $proc_id})
    MERGE (cap)-[:CONTAINS {
      created_at: datetime()
    }]->(proc)
    """
    
    # 2. 流程被系统支撑
    query = """
    MATCH (proc:BusinessProcess {uuid: $proc_id})
    MATCH (sys:ApplicationSystem {uuid: $sys_id})
    MERGE (proc)-[:SUPPORTED_BY {
      support_type: $support_type,
      coverage: $coverage,
      critical: $critical,
      created_at: datetime()
    }]->(sys)
    """
    
    # 3. 系统被AI服务增强
    query = """
    MATCH (sys:ApplicationSystem {uuid: $sys_id})
    MATCH (service:AIService {uuid: $service_id})
    MERGE (sys)-[:ENHANCED_BY {
      enhancement: $enhancement,
      expected_improvement: $expected_improvement,
      created_at: datetime()
    }]->(service)
    """
    
    # 4. 平台提供AI服务
    query = """
    MATCH (platform:AIPlatform {uuid: $platform_id})
    MATCH (service:AIService {uuid: $service_id})
    MERGE (platform)-[:PROVIDES {
      created_at: datetime()
    }]->(service)
    """

    async def _link_process_to_capability(

        for process in processes:

                # 在Neo4j中创建SUPPORTED_BY关系
                await self.neo4j.create_relationship(
                    source_id=process.neo4j_node_id,

    async def _enhance_systems_with_platform(self) -> List[Dict]:
        """用平台增强系统"""
        # 获取所有有痛点的业务流程
        processes = self.db.query(BusinessProcess).filter(
            BusinessProcess.meta_data["pain_points"].isnot(None)
        ).all()
        
        # 获取平台
        platform = self.db.query(AIPlatform).filter(
            AIPlatform.name == "LuminaOS"
        ).first()
        
        if not platform:
            return []
        
        enhancements = []
        
        for process in processes:
            pain_points = process.meta_data.get("pain_points", [])
            
            for pain_point in pain_points:
                # 根据痛点类型创建AI服务
                service_name = f"{process.name}智能{self._get_enhancement_type(pain_point)}"
                
                ai_service = AIService(
                    platform_id=platform.id,
                    name=service_name,
                    service_type="流程自动化",
                    pain_point=pain_point,
                    expected_improvement="效率提升40%",
                    status="active",
                    meta_data={
                        "process_id": str(process.id),
                        "process_name": process.name
                    }
                )
                self.db.add(ai_service)
                self.db.flush()
                
                # 创建Neo4j节点
                service_neo4j_id = await self.graph_service.create_ai_service_node(ai_service)
                
                # 创建平台提供服务的关系
                if platform.neo4j_node_id:
                    await self.neo4j.create_relationship(
                        source_id=platform.neo4j_node_id,
                        target_id=service_neo4j_id,
                        rel_type="PROVIDES",
                        properties={"created_at": datetime.utcnow().isoformat()}
                    )
                
                # 获取支撑该流程的系统
                sap_system = self.db.query(ApplicationSystem).filter(
                    ApplicationSystem.name == "SAP ERP"
                ).first()
                
                if sap_system and sap_system.neo4j_node_id:
                    # 创建系统被增强的关系
                    await self.neo4j.create_relationship(
                        source_id=sap_system.neo4j_node_id,
                        target_id=service_neo4j_id,
                        rel_type="ENHANCED_BY",
                        properties={
                            "enhancement": service_name,
                            "expected_improvement": "效率提升40%",
                            "created_at": datetime.utcnow().isoformat()
                        }
                    )
                    
                    enhancements.append({
                        "system": sap_system.name,
                        "service": service_name,
                        "pain_point": pain_point,
                        "type": "ENHANCED_BY"
                    })
        
        self.db.commit()
        return enhancements
    
    def _get_enhancement_type(self, pain_point: str) -> str:
        """根据痛点获取增强类型"""
        if "审批" in pain_point:
            return "审批"
        elif "对账" in pain_point or "核算" in pain_point:
            return "分析"
        elif "管理" in pain_point:
            return "管理"
        else:
            return "优化"

    db: Session = Depends(get_db),
    neo4j_client: Neo4jClient = Depends(get_neo4j_client)
):
    """分析平台对业务的价值"""
    service = ChemicalIndustryModelingService(db, neo4j_client)
    
    # 查询：哪些业务痛点被AI服务解决
    query = """
    MATCH (proc:BusinessProcess)-[:SUPPORTED_BY]->(sys:ApplicationSystem)
    MATCH (sys)-[:ENHANCED_BY]->(service:AIService)
    MATCH (service)<-[:PROVIDES]-(platform:AIPlatform)
    RETURN proc.name as process,
           proc.pain_points as pain_points,
           service.name as ai_service,
           service.expected_improvement as improvement
    """
    
    result = await neo4j_client.execute_query(query)
    return {"value_analysis": result}

@router.get("/impact-analysis/{process_id}")
async def analyze_process_impact(
    process_id: str,
    db: Session = Depends(get_db),
    neo4j_client: Neo4jClient = Depends(get_neo4j_client)
):
    """分析业务流程的影响范围"""
    process = db.query(BusinessProcess).filter(
        BusinessProcess.id == UUID(process_id)
    ).first()
    
    if not process or not process.neo4j_node_id:
        raise HTTPException(404, "流程不存在或未同步到Neo4j")
    
    # 查询影响范围
    query = """
    MATCH (proc:BusinessProcess {uuid: $process_id})
    OPTIONAL MATCH (proc)-[:SUPPORTED_BY]->(sys:ApplicationSystem)
    OPTIONAL MATCH (sys)-[:ENHANCED_BY]->(service:AIService)
    RETURN proc,
           collect(DISTINCT sys) as systems,
           collect(DISTINCT service) as ai_services
    """
    
    result = await neo4j_client.execute_query(
        query, {"process_id": process_id}
    )
    
    return {
        "process": {
            "id": str(process.id),
            "name": process.name,
            "pain_points": process.meta_data.get("pain_points", [])
        },
        "impact": result[0] if result else {}
    }
