#### KnowledgeGraphView.tsx

```typescript
import React, { useState, useEffect } from 'react';
import ForceGraph2D from 'react-force-graph-2d';
import { GraphVisualization } from './GraphVisualization';
import { GraphControls } from './GraphControls';
import { GraphSidebar } from './GraphSidebar';
import { knowledgeGraphService } from '@/services/knowledgeGraphService';

export const KnowledgeGraphView: React.FC = () => {
  const [graphData, setGraphData] = useState({ nodes: [], links: [] });
  const [selectedNode, setSelectedNode] = useState(null);
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    loadGraphData();
  }, []);

  const loadGraphData = async () => {
    setLoading(true);
    try {
      const data = await knowledgeGraphService.getGraphData();
      setGraphData(data);
    } catch (error) {
      console.error('Failed to load graph data:', error);
    } finally {
      setLoading(false);
    }
  };

  const handleNodeClick = (node: any) => {
    setSelectedNode(node);
  };

  return (
    <div className="knowledge-graph-view">
      <GraphControls 
        onRefresh={loadGraphData}
        onSearch={(query) => {/* 搜索逻辑 */}}
      />
      <div className="graph-container">
        <GraphVisualization
          data={graphData}
          onNodeClick={handleNodeClick}
          loading={loading}
        />
        <GraphSidebar
          selectedNode={selectedNode}
          onClose={() => setSelectedNode(null)}
        />
      </div>
    </div>
  );
};
```

#### GraphVisualization.tsx

```typescript
import React, { useRef, useCallback } from 'react';
import ForceGraph2D from 'react-force-graph-2d';

interface GraphVisualizationProps {
  data: { nodes: any[]; links: any[] };
  onNodeClick: (node: any) => void;
  loading: boolean;
}

export const GraphVisualization: React.FC<GraphVisualizationProps> = ({
  data,
  onNodeClick,
  loading
}) => {
  const graphRef = useRef<any>();

  const handleNodeClick = useCallback((node: any) => {
    onNodeClick(node);
  }, [onNodeClick]);

  if (loading) {
    return <div>Loading graph...</div>;
  }

  return (
    <ForceGraph2D
      ref={graphRef}
      graphData={data}
      nodeLabel={(node: any) => node.name || node.id}
      nodeColor={(node: any) => node.color || '#1f77b4'}
      linkColor={() => 'rgba(0,0,0,0.2)'}
      linkWidth={2}
      onNodeClick={handleNodeClick}
      nodeCanvasObject={(node: any, ctx: CanvasRenderingContext2D) => {
        const label = node.name || node.id;
        const fontSize = 12;
        ctx.font = `${fontSize}px Sans-Serif`;
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        ctx.fillStyle = node.color || '#1f77b4';
        ctx.fillText(label, node.x, node.y);
      }}
    />
  );
};
```

#### knowledgeGraphService.ts

```typescript
import axios from 'axios';

const API_BASE_URL = process.env.REACT_APP_API_GATEWAY_URL || 'http://localhost:8080';

export const knowledgeGraphService = {
  // 获取图谱数据
  async getGraphData(): Promise<{ nodes: any[]; links: any[] }> {
    const response = await axios.get(`${API_BASE_URL}/api/knowledge-graph/graph`);
    return response.data;
  },

  // 获取实体列表
  async getEntities(params?: any): Promise<any[]> {
    const response = await axios.get(`${API_BASE_URL}/api/knowledge-graph/entities`, { params });
    return response.data;
  },

  // 获取关系列表
  async getRelations(params?: any): Promise<any[]> {
    const response = await axios.get(`${API_BASE_URL}/api/knowledge-graph/relations`, { params });
    return response.data;
  },

  // 获取实体详情
  async getEntity(id: string): Promise<any> {
    const response = await axios.get(`${API_BASE_URL}/api/knowledge-graph/entities/${id}`);
    return response.data;
  },

  // 查询图谱
  async queryGraph(query: string): Promise<any> {
    const response = await axios.post(`${API_BASE_URL}/api/knowledge-graph/query`, { query });
    return response.data;
  }
};
```

### 2.2 ComponentShowcase.tsx

```typescript
import React, { useState } from 'react';
import { ComponentCard } from '@/components/showcase/ComponentCard';
import { SearchInput } from '@/components/common/SearchInput';

const components = [
  {
    name: 'Button',
    category: '基础组件',
    description: '按钮组件',
    preview: <Button>示例</Button>,
    path: '/components/button'
  },
  {
    name: 'Input',
    category: '基础组件',
    description: '输入框组件',
    preview: <Input placeholder="示例" />,
    path: '/components/input'
  },
  // ... 更多组件
];

export const ComponentShowcase: React.FC = () => {
  const [searchQuery, setSearchQuery] = useState('');
  const [selectedCategory, setSelectedCategory] = useState('all');

  const filteredComponents = components.filter(comp => {
    const matchesSearch = comp.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
                         comp.description.toLowerCase().includes(searchQuery.toLowerCase());
    const matchesCategory = selectedCategory === 'all' || comp.category === selectedCategory;
    return matchesSearch && matchesCategory;
  });

  const categories = ['all', ...new Set(components.map(c => c.category))];

  return (
    <div className="component-showcase">
      <div className="showcase-header">
        <h1>组件库</h1>
        <SearchInput
          value={searchQuery}
          onChange={setSearchQuery}
          placeholder="搜索组件..."
        />
        <CategoryFilter
          categories={categories}
          selected={selectedCategory}
          onChange={setSelectedCategory}
        />
      </div>
      <div className="component-grid">
        {filteredComponents.map(comp => (
          <ComponentCard key={comp.name} component={comp} />
        ))}
      </div>
    </div>
  );
};
```

## 四、知识库详细页面实施

### 4.1 KnowledgeBaseDetail.tsx

```typescript
import React, { useState, useEffect } from 'react';
import { useParams } from 'react-router-dom';
import { knowledgeBaseService } from '@/services/knowledgeBaseService';
import { DocumentList } from '@/components/DocumentList';
import { StatCard } from '@/components/charts/StatCard';

export const KnowledgeBaseDetail: React.FC = () => {
  const { id } = useParams<{ id: string }>();
  const [kb, setKb] = useState(null);
  const [documents, setDocuments] = useState([]);
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    if (id) {
      loadKnowledgeBase(id);
    }
  }, [id]);

  const loadKnowledgeBase = async (kbId: string) => {
    setLoading(true);
    try {
      const [kbData, docsData] = await Promise.all([
        knowledgeBaseService.getKnowledgeBase(kbId),
        knowledgeBaseService.getDocuments(kbId)
      ]);
      setKb(kbData);
      setDocuments(docsData);
    } catch (error) {
      console.error('Failed to load knowledge base:', error);
    } finally {
      setLoading(false);
    }
  };

  if (loading) {
    return <div>Loading...</div>;
  }

  if (!kb) {
    return <div>Knowledge base not found</div>;
  }

  return (
    <div className="knowledge-base-detail">
      <div className="kb-header">
        <h1>{kb.name}</h1>
        <p>{kb.description}</p>
      </div>

      <div className="kb-stats">
        <StatCard title="文档数" value={kb.document_count} />
        <StatCard title="总Chunks" value={kb.total_chunks} />
        <StatCard title="状态" value={kb.status} />
      </div>

      <div className="kb-documents">
        <h2>文档列表</h2>
        <DocumentList documents={documents} />
      </div>
    </div>
  );
};
```
