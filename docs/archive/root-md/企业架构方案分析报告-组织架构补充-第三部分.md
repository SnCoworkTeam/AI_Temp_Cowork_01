## 📊 三、使用场景

### 3.1 场景1: 组织责任查询

```python
# API: 查询组织负责的业务
GET /api/enterprise-architecture/organizations/{org_id}/responsibilities

# Neo4j查询
MATCH (org:OrganizationUnit {uuid: $org_id})
OPTIONAL MATCH (org)-[:OWNS]->(cap:BusinessCapability)
OPTIONAL MATCH (org)-[:EXECUTES]->(proc:BusinessProcess)
OPTIONAL MATCH (org)-[:USES]->(sys:ApplicationSystem)
RETURN 
  org.name as organization,
  collect(DISTINCT cap.name) as capabilities,
  collect(DISTINCT proc.name) as processes,
  collect(DISTINCT sys.name) as systems
```

### 3.2 场景2: 组织资源统计

```python
# API: 统计组织的IT资源使用
GET /api/enterprise-architecture/organizations/{org_id}/resources

# Neo4j查询
MATCH (org:OrganizationUnit {uuid: $org_id})
OPTIONAL MATCH (org)-[:USES]->(sys:ApplicationSystem)
OPTIONAL MATCH (org)-[:USES]->(service:AIService)
RETURN 
  org.name as organization,
  count(DISTINCT sys) as system_count,
  count(DISTINCT service) as ai_service_count,
  collect(DISTINCT sys.name) as systems,
  collect(DISTINCT service.name) as ai_services
```

### 3.3 场景3: 组织变革影响分析

```python
# API: 分析组织合并的影响
POST /api/enterprise-architecture/organizations/merge-impact
{
  "source_org_ids": ["org1", "org2"],
  "target_org_id": "org3"
}

# Neo4j查询
MATCH (org1:OrganizationUnit {uuid: $org1_id})-[:USES]->(sys:ApplicationSystem)
MATCH (org2:OrganizationUnit {uuid: $org2_id})-[:USES]->(sys)
RETURN 
  sys.name as shared_system,
  "需要整合" as action

MATCH (org1:OrganizationUnit {uuid: $org1_id})-[:USES]->(sys1:ApplicationSystem)
MATCH (org2:OrganizationUnit {uuid: $org2_id})-[:USES]->(sys2:ApplicationSystem)
WHERE sys1 <> sys2
RETURN 
  collect(DISTINCT sys1.name) + collect(DISTINCT sys2.name) as systems_to_integrate
```

## 🔧 四、服务层实现

### 4.1 组织架构服务

```python
# metadata-service/src/services/organization_architecture_service.py

class OrganizationArchitectureService:
    """组织架构服务"""
    
    def __init__(self, db: Session, neo4j_client: Neo4jClient):
        self.db = db
        self.neo4j = neo4j_client
        self.graph_service = OrganizationArchitectureGraphService(db, neo4j_client)
    
    async def get_organization_responsibilities(self, org_id: UUID) -> Dict:
        """获取组织的责任范围"""
        org = self.db.query(OrganizationUnit).filter(
            OrganizationUnit.id == org_id
        ).first()
        
        if not org:
            raise ValueError(f"组织不存在: {org_id}")
        
        # 查询业务能力
        capabilities = self.db.query(BusinessCapability).join(
            OrganizationBusinessRelationship
        ).filter(
            OrganizationBusinessRelationship.organization_id == org_id,
            OrganizationBusinessRelationship.entity_type == "BusinessCapability",
            OrganizationBusinessRelationship.relationship_type == "OWNS"
        ).all()
        
        # 查询业务流程
        processes = self.db.query(BusinessProcess).join(
            OrganizationBusinessRelationship
        ).filter(
            OrganizationBusinessRelationship.organization_id == org_id,
            OrganizationBusinessRelationship.entity_type == "BusinessProcess",
            OrganizationBusinessRelationship.relationship_type == "EXECUTES"
        ).all()
        
        # 查询应用系统
        systems = self.db.query(ApplicationSystem).join(
            OrganizationBusinessRelationship
        ).filter(
            OrganizationBusinessRelationship.organization_id == org_id,
            OrganizationBusinessRelationship.entity_type == "ApplicationSystem",
            OrganizationBusinessRelationship.relationship_type == "USES"
        ).all()
        
        return {
            "organization": {
                "id": str(org.id),
                "name": org.name,
                "code": org.code
            },
            "capabilities": [self._capability_to_dict(c) for c in capabilities],
            "processes": [self._process_to_dict(p) for p in processes],
            "systems": [self._system_to_dict(s) for s in systems]
        }
    
    async def analyze_organization_resources(self, org_id: UUID) -> Dict:
        """分析组织的IT资源使用"""
        # 使用Neo4j查询
        query = """
        MATCH (org:OrganizationUnit {uuid: $org_id})
        OPTIONAL MATCH (org)-[:USES]->(sys:ApplicationSystem)
        OPTIONAL MATCH (org)-[:USES]->(service:AIService)
        RETURN 
          org.name as organization,
          count(DISTINCT sys) as system_count,
          count(DISTINCT service) as ai_service_count,
          collect(DISTINCT sys.name) as systems,
          collect(DISTINCT service.name) as ai_services
        """
        
        result = await self.neo4j.execute_query(query, {"org_id": str(org_id)})
        
        return result[0] if result else {}
```
