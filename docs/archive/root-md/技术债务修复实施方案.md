```python
async def update_document(
    self,
    document_id: str,
    title: Optional[str] = None,
    category: Optional[str] = None,
    tags: Optional[List[str]] = None,
    knowledge_base_id: Optional[str] = None,
    metadata: Optional[Dict[str, Any]] = None
) -> Dict[str, Any]:
    """更新文档"""
    try:
        # 获取文档
        document = self.document_repo.get_by_id(document_id)
        if not document:
            raise ValueError(f"Document {document_id} not found")
        
        # 构建更新数据
        updates = {}
        if title is not None:
            updates['title'] = title
        if category is not None:
            updates['category'] = category
        if tags is not None:
            updates['tags'] = tags
        if knowledge_base_id is not None:
            try:
                updates['knowledge_base_id'] = UUID(knowledge_base_id)
            except ValueError:
                raise ValueError(f"Invalid knowledge_base_id: {knowledge_base_id}")
        if metadata is not None:
            updates['metadata'] = metadata
        
        # 更新文档
        updated_doc = self.document_repo.update_document(document_id, **updates)
        self.db.commit()
        
        return {
            "document_id": str(updated_doc.id),
            "message": "Document updated successfully"
        }
        
    except Exception as e:
        self.db.rollback()
        logger.error(f"Error updating document: {str(e)}", exc_info=True)
        raise
```

@router.post(
    "/search/vector",
    summary="向量搜索",
    description="基于向量相似度的文档搜索",
    tags=["Search"]
)
async def vector_search(
    request: VectorSearchRequest,
    db: Session = Depends(get_db)
) -> Dict[str, Any]:

            query=request.query,

### 2.2 文档处理优化

#### 优化方案

**步骤1: 添加处理超时**

```python
# 在文档处理服务中添加
PROCESSING_TIMEOUT = 30 * 60  # 30分钟

async def _process_document_with_timeout(self, document_id: str):
    """带超时的文档处理"""
    try:
        await asyncio.wait_for(
            self._process_document(document_id),
            timeout=PROCESSING_TIMEOUT
        )
    except asyncio.TimeoutError:
        logger.error(f"Document {document_id} processing timeout")
        self.document_repo.update_status(document_id, "failed")
        # 发送告警
```

**步骤2: 实现重试机制**

```python
MAX_RETRIES = 3
RETRY_DELAY = 60  # 1分钟

async def _process_document_with_retry(self, document_id: str):
    """带重试的文档处理"""
    for attempt in range(MAX_RETRIES):
        try:
            await self._process_document(document_id)
            return
        except Exception as e:
            if attempt < MAX_RETRIES - 1:
                logger.warning(f"Document {document_id} processing failed, retrying...")
                await asyncio.sleep(RETRY_DELAY)
            else:
                logger.error(f"Document {document_id} processing failed after {MAX_RETRIES} attempts")
                self.document_repo.update_status(document_id, "failed")
                raise
```
