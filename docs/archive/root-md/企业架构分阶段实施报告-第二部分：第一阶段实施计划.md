#### 2.3.2 技术实例模型设计

```python
class TechnologyInstance(BaseModel, TimestampMixin):
    """技术实例 - 按系统分类"""
    __tablename__ = "technology_instances"
    
    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    name = Column(String(255), nullable=False, index=True)  # "Oracle-SAP-ERP-001"
    instance_id = Column(String(255), nullable=True, unique=True, index=True)
    
    # 归属关系
    application_system_id = Column(UUID(as_uuid=True), ForeignKey("application_systems.id"), nullable=True, index=True)
    owner_department = Column(String(255), nullable=True)
    owner_team = Column(String(255), nullable=True)
    
    # 技术属性
    technology_type = Column(String(100), nullable=False, index=True)  # Database, Middleware等
    technology_name = Column(String(255), nullable=False, index=True)  # Oracle, Redis等
    technology_type_id = Column(UUID(as_uuid=True), ForeignKey("technology_types.id"), nullable=True)
    vendor = Column(String(255), nullable=True)
    version = Column(String(50), nullable=True)
    
    # 部署属性
    deployment_type = Column(String(50), nullable=True, default="independent")
    host = Column(String(255), nullable=True)
    port = Column(Integer, nullable=True)
    location = Column(String(255), nullable=True)
    capacity = Column(String(100), nullable=True)
    
    # 状态
    status = Column(String(50), nullable=True, default="active")
    lifecycle_status = Column(String(50), nullable=True)
    
    neo4j_node_id = Column(String(255), nullable=True)
    meta_data = Column(JSONB, nullable=True, default=dict)
```
