```python
# scripts/import_g437_to_enterprise_architecture.py

import asyncio
from docx import Document
import re
from pathlib import Path
from typing import Dict, List
import sys

# 添加项目路径
project_root = Path(__file__).parent.parent
sys.path.insert(0, str(project_root))

from database.src.core.session import get_db
from database.src.core.neo4j_client import Neo4jClient, get_neo4j_settings
from metadata_service.src.services.chemical_industry_modeling_service import (
    ChemicalIndustryModelingService

async def parse_g437_document(file_path: str) -> Dict:
    """解析G437文档，提取架构信息"""
    doc = Document(file_path)
    
    architecture_data = {
        "business_processes": [],
        "application_systems": [],
        "data_entities": [],
        "relationships": []
    }
    
    current_section = None
    current_process = None
    
    for paragraph in doc.paragraphs:
        text = paragraph.text.strip()
        
        if not text:
            continue
        
        # 识别章节
        if "业务流程" in text or "业务架构" in text:
            current_section = "business"
        elif "应用系统" in text or "应用架构" in text:
            current_section = "application"
        elif "数据实体" in text or "数据架构" in text:
            current_section = "data"
        elif "ERP" in text.upper() or "SAP" in text.upper():
            current_section = "application"
        
        # 提取业务流程
        if current_section == "business":
            # 匹配流程编号和名称
            process_match = re.match(r'(\d+\.\d+)\s+(.+)', text)
            if process_match:
                current_process = {
                    "code": process_match.group(1),
                    "name": process_match.group(2),
                    "description": text,
                    "pain_points": []
                }
                architecture_data["business_processes"].append(current_process)
            elif current_process and ("问题" in text or "痛点" in text or "困难" in text):
                # 提取痛点
                pain_points = re.findall(r'[，,]\s*([^，,]+)', text)
                current_process["pain_points"].extend(pain_points)
        
        # 提取应用系统
        if current_section == "application":
            if "SAP" in text or "ERP" in text:
                architecture_data["application_systems"].append({
                    "name": "SAP ERP",
                    "type": "核心系统",
                    "vendor": "SAP",
                    "description": text
                })
            elif "系统" in text and len(text) < 50:
                system_name = text.replace("系统", "").strip()
                architecture_data["application_systems"].append({
                    "name": system_name,
                    "type": "专业系统",
                    "vendor": "未知"
                })
    
    return architecture_data

async def import_g437_to_architecture(file_path: str):
    """将G437文档导入到企业架构"""
    # 解析文档
    data = await parse_g437_document(file_path)
    
    # 获取数据库会话
    db = next(get_db())
    
    # 获取Neo4j客户端
    neo4j_settings = get_neo4j_settings()
    neo4j_client = Neo4jClient(neo4j_settings)
    await neo4j_client.connect()
    
    try:
        # 创建建模服务
        service = ChemicalIndustryModelingService(db, neo4j_client)
        
        # 导入业务流程
        for process_data in data["business_processes"]:
            process = BusinessProcess(
                name=process_data["name"],
                description=process_data.get("description"),
                owner="待定",
                status="active",
                classification="化工行业",
                meta_data={
                    "source": "G437文档",
                    "code": process_data.get("code"),
                    "pain_points": process_data.get("pain_points", []),
                    "industry": "chemical"
                }
            )
            db.add(process)
            db.flush()
            
            # 创建Neo4j节点
            await service.graph_service.create_business_process_node(process)
        
        # 导入应用系统
        for system_data in data["application_systems"]:
            system = ApplicationSystem(
                name=system_data["name"],
                system_type=system_data["type"],
                vendor=system_data.get("vendor"),
                status="active",
                meta_data={
                    "source": "G437文档",
                    "industry": "chemical"
                }
            )
            db.add(system)
            db.flush()
            
            # 创建Neo4j节点
            await service.graph_service.create_application_system_node(system)
        
        db.commit()
        print(f"✅ 成功导入 {len(data['business_processes'])} 个业务流程")
        print(f"✅ 成功导入 {len(data['application_systems'])} 个应用系统")
        
    finally:
        await neo4j_client.disconnect()
        db.close()

if __name__ == "__main__":
```

```python
# scripts/import_enterprise_architecture_from_excel.py

import pandas as pd
import asyncio
from typing import Dict, List
from database.src.core.session import get_db
from database.src.core.neo4j_client import Neo4jClient, get_neo4j_settings
from metadata_service.src.services.chemical_industry_modeling_service import (
    ChemicalIndustryModelingService
)

async def import_from_excel(file_path: str):
    """从Excel导入企业架构数据"""
    db = next(get_db())
    neo4j_settings = get_neo4j_settings()
    neo4j_client = Neo4jClient(neo4j_settings)
    await neo4j_client.connect()
    
    try:
        service = ChemicalIndustryModelingService(db, neo4j_client)
        
        # 1. 导入业务能力
        capabilities_df = pd.read_excel(file_path, sheet_name='业务能力')
        for _, row in capabilities_df.iterrows():
            capability = BusinessCapability(
                name=row['能力名称'],
                description=row.get('能力描述'),
                level=row.get('层级', 1),
                meta_data={"industry": row.get('行业分类', 'chemical')}
            )
            db.add(capability)
            db.flush()
            await service.graph_service.create_business_capability_node(capability)
        
        # 2. 导入业务流程
        processes_df = pd.read_excel(file_path, sheet_name='业务流程')
        for _, row in processes_df.iterrows():
            pain_points = str(row.get('业务痛点', '')).split(',') if pd.notna(row.get('业务痛点')) else []
            sap_modules = str(row.get('SAP模块', '')).split(',') if pd.notna(row.get('SAP模块')) else []
            
            process = BusinessProcess(
                name=row['流程名称'],
                description=row.get('流程描述'),
                owner=row.get('负责人'),
                status="active",
                classification="化工行业",
                meta_data={
                    "code": row.get('流程编码'),
                    "criticality": row.get('关键性', '中'),
                    "pain_points": [p.strip() for p in pain_points if p.strip()],
                    "sap_modules": [m.strip() for m in sap_modules if m.strip()],
                    "industry": "chemical"
                }
            )
            db.add(process)
            db.flush()
            await service.graph_service.create_business_process_node(process)
            
            # 关联到业务能力
            if pd.notna(row.get('关联能力')):
                capability_name = row['关联能力']
                capability = db.query(BusinessCapability).filter(
                    BusinessCapability.name == capability_name
                ).first()
                if capability and capability.neo4j_node_id and process.neo4j_node_id:
                    await neo4j_client.create_relationship(
                        source_id=capability.neo4j_node_id,
                        target_id=process.neo4j_node_id,
                        rel_type="CONTAINS"
                    )
        
        # 3. 导入应用系统
        systems_df = pd.read_excel(file_path, sheet_name='应用系统')
        for _, row in systems_df.iterrows():
            system = ApplicationSystem(
                name=row['系统名称'],
                system_type=row.get('系统类型'),
                vendor=row.get('供应商'),
                version=row.get('版本'),
                status=row.get('状态', 'active'),
                meta_data={"industry": "chemical"}
            )
            db.add(system)
            db.flush()
            await service.graph_service.create_application_system_node(system)
            
            # 关联到业务流程
            if pd.notna(row.get('支撑的流程')):
                process_names = str(row['支撑的流程']).split(',')
                for proc_name in process_names:
                    process = db.query(BusinessProcess).filter(
                        BusinessProcess.name == proc_name.strip()
                    ).first()
                    if process and process.neo4j_node_id and system.neo4j_node_id:
                        await neo4j_client.create_relationship(
                            source_id=process.neo4j_node_id,
                            target_id=system.neo4j_node_id,
                            rel_type="SUPPORTED_BY",
                            properties={
                                "support_type": "主要支撑",
                                "coverage": "80%",
                                "critical": True
                            }
                        )
        
        db.commit()
        print("✅ Excel导入完成")
        
    finally:
        await neo4j_client.disconnect()
        db.close()
```

#### 2.3.3 改进机会发现查询

```cypher
// 查询：哪些业务痛点还没有AI解决方案
MATCH (proc:BusinessProcess)
WHERE proc.industry = 'chemical'
  AND size(proc.pain_points) > 0
OPTIONAL MATCH (proc)-[:SUPPORTED_BY]->(sys)-[:ENHANCED_BY]->(service)
WITH proc, collect(DISTINCT service.pain_point) as solved_pains
WHERE NOT ALL(pain IN proc.pain_points WHERE pain IN solved_pains)
RETURN 
  proc.name as process,
  proc.pain_points as all_pains,
  solved_pains,
  [pain IN proc.pain_points WHERE NOT pain IN solved_pains] as unsolved_pains
ORDER BY size(unsolved_pains) DESC
```

```python
# metadata-service/src/services/architecture_data_validation.py

class ArchitectureDataValidationService:
    """架构数据验证服务"""
    
    def validate_business_process(self, process: BusinessProcess) -> Dict:
        """验证业务流程数据"""
        errors = []
        warnings = []
        
        # 必填字段检查
        if not process.name:
            errors.append("流程名称不能为空")
        
        # 痛点检查
        if not process.meta_data.get("pain_points"):
            warnings.append("建议记录业务痛点，以便识别改进机会")
        
        # 关键性检查
        if not process.meta_data.get("criticality"):
            warnings.append("建议标识流程关键性")
        
        # SAP模块关联检查
        if not process.meta_data.get("sap_modules"):
            warnings.append("建议关联SAP模块，便于系统映射")
        
        return {
            "valid": len(errors) == 0,
            "errors": errors,
            "warnings": warnings
        }
    
    def validate_relationship(self, source_type: str, target_type: str, rel_type: str) -> bool:
        """验证关系是否合理"""
        valid_relationships = {
            ("BusinessCapability", "BusinessProcess", "CONTAINS"),
            ("BusinessProcess", "ApplicationSystem", "SUPPORTED_BY"),
            ("ApplicationSystem", "AIService", "ENHANCED_BY"),
            ("AIPlatform", "AIService", "PROVIDES")
        }
        
        return (source_type, target_type, rel_type) in valid_relationships
```

```python
                AIService.pain_point.isnot(None)

    """
    result = await neo4j_client.execute_query(query)
    checks["business_processes"]["with_systems"] = result[0]["procs_with_systems"] if result else 0
```
