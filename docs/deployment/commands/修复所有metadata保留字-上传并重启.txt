# 2. workflow_models.py - WorkflowDefinition.metadata -> workflow_metadata, WorkflowExecution.metadata -> execution_metadata

# 4. knowledge_models.py - Document.metadata -> doc_metadata, DocumentChunk.metadata -> chunk_metadata
