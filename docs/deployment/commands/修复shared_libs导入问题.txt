# 修复shared_libs导入问题
# 问题：ModuleNotFoundError: No module named 'shared_libs'

# ============================================
# 第一步：检查服务器上的shared_libs目录
# ============================================
# 在服务器上执行：
cd /opt/enterprise-ai-platform

# 检查目录是否存在
ls -la shared_libs/ 2>&1

# 检查关键文件是否存在
ls -la shared_libs/__init__.py
ls -la shared_libs/schemas/monitoring_schemas.py
ls -la shared_libs/common/logger.py

# 检查容器内的挂载
sudo docker exec enterprise-ai-mcp-gateway ls -la /shared_libs 2>&1

# ============================================
# 第二步：如果目录不存在，需要上传shared_libs
# ============================================
# 在本地PowerShell执行上传命令（见"上传shared_libs目录.txt"）

# ============================================
# 第三步：重启服务
# ============================================
sudo docker compose restart mcp-gateway
sleep 20
sudo docker compose logs --tail=30 mcp-gateway

# ============================================
# 一键检查命令（服务器上）
# ============================================
cd /opt/enterprise-ai-platform && echo "=== Local shared_libs ===" && ls -la shared_libs/ 2>&1 | head -10 && echo "" && echo "=== Container shared_libs ===" && sudo docker exec enterprise-ai-mcp-gateway ls -la /shared_libs 2>&1 | head -10 && echo "" && echo "=== Test Import ===" && sudo docker exec enterprise-ai-mcp-gateway python -c "import sys; sys.path.insert(0, '/shared_libs'); import shared_libs; print('shared_libs imported OK')" 2>&1
