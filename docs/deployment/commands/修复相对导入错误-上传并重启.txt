# 第一步：在本地PowerShell上传修复后的main.py
# ============================================
# 在本地Windows PowerShell中执行：

scp -i "E:\enterprise-ai-platform\enterprise_ai_platform.pem" -o StrictHostKeyChecking=no mcp-gateway/src/main.py ubuntu@43.143.139.197:/opt/enterprise-ai-platform/mcp-gateway/src/

# ============================================
# 第二步：在服务器上重启服务（代码已热重载）
# ============================================
# SSH到服务器后执行：

cd /opt/enterprise-ai-platform

# 1. 重启mcp-gateway（代码已热重载，但需要重启确保加载新代码）
sudo docker compose restart mcp-gateway

# 2. 等待启动
sleep 20

# 3. 检查状态
sudo docker compose ps mcp-gateway

sudo docker compose logs --tail=30 mcp-gateway

# 5. 测试服务
curl http://localhost:8001/api/health
