# 永久解决方案 - 统一使用docker compose管理所有服务
# 这样可以避免postgres被停止的问题

cd /opt/enterprise-ai-platform

# ============================================
# 步骤1: 停止所有不在compose管理下的容器
# ============================================
# 停止postgres（如果正在运行）
sudo docker stop enterprise-ai-postgres

# ============================================
# 步骤2: 使用docker compose统一启动所有服务
# ============================================
# 这样所有服务都在docker compose的管理下，不会互相影响
sudo docker compose up -d postgres redis mcp-gateway

# 等待所有服务启动
sleep 20

# ============================================
# 步骤3: 验证所有服务状态
# ============================================
sudo docker compose ps

# 应该看到所有服务都是 "Up" 状态

# ============================================
# 步骤4: 查看服务日志
# ============================================
sudo docker compose logs --tail=30 mcp-gateway

# ============================================
# 步骤5: 测试服务
# ============================================
curl http://localhost:8001/api/health

# ============================================
# 以后启动服务的正确方法
# ============================================
# 方法1: 启动所有服务
sudo docker compose up -d

# 方法2: 启动特定服务（不会影响其他服务）
sudo docker compose up -d mcp-gateway

# 方法3: 重启特定服务
sudo docker compose restart mcp-gateway

# ============================================
# 停止服务
# ============================================
# 停止所有服务
sudo docker compose down

# 停止特定服务（不会影响其他服务）
sudo docker compose stop mcp-gateway

# ============================================
# 一键修复命令
# ============================================
sudo docker stop enterprise-ai-postgres 2>/dev/null; sudo docker compose up -d postgres redis mcp-gateway && sleep 20 && sudo docker compose ps && echo "" && curl http://localhost:8001/api/health
