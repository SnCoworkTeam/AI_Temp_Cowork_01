# 验证mcp-gateway修复成功 - 最终检查
# 在服务器上执行以下命令

cd /opt/enterprise-ai-platform

# 1. 等待健康检查完成（约30秒）
echo "Waiting for health check to complete..."
sleep 30

# 2. 检查容器状态（应该是 healthy，不是 unhealthy）
echo ""
echo "=== Container Status ==="
sudo docker compose ps mcp-gateway

# 或者
sudo docker ps | grep mcp-gateway

# 3. 查看最新日志（应该没有导入错误了）
echo ""
echo "=== Recent Logs ==="
sudo docker compose logs --tail=30 mcp-gateway

# 4. 检查是否有shared_libs导入错误（应该没有）
echo ""
echo "=== Checking for shared_libs errors (should be empty) ==="
sudo docker compose logs mcp-gateway | grep -i "shared_libs\|ModuleNotFoundError" | tail -5

# 5. 检查是否有LOG_LEVEL错误（应该没有）
echo ""
echo "=== Checking for LOG_LEVEL errors (should be empty) ==="
sudo docker compose logs mcp-gateway | grep -i "log-level\|Invalid value" | tail -5

# 6. 检查是否有启动成功消息
echo ""
echo "=== Checking for startup success ==="
sudo docker compose logs mcp-gateway | grep -i "startup\|application\|uvicorn\|started\|running" | tail -5

# 7. 测试服务健康检查端点
echo ""
echo "=== Testing Health Endpoint ==="
curl -v http://localhost:8001/api/health

# 8. 测试服务根路径
echo ""
echo "=== Testing Root Path ==="
curl -v http://localhost:8001/
