# 在服务器上直接修复shared_libs/__init__.py
# 复制以下命令到服务器执行

cd /opt/enterprise-ai-platform

# 备份原文件
cp shared_libs/__init__.py shared_libs/__init__.py.bak

# 创建修复后的文件
cat > shared_libs/__init__.py << 'EOF'
"""共享库包"""
# 确保包可以被导入，即使子模块导入失败
__all__ = []

# 导出元数据模型（如果存在）
try:
    from .src.models import (
        DataAssetMetadata,
        AIModelMetadata,
        BusinessEntityMetadata,
        WorkflowMetadata,
        DataLineageRelation,
        QualityMetrics,
        MetadataSearchResult,
        MetadataFilter,
        MetadataResponse,
        TechnicalMetadata,
        BusinessMetadata,
        OperationalMetadata,
        DataAssetType,
        ModelType,
        EntityType,
        WorkflowType,
        LineageRelationType,
        QualityLevel,
        QualityScore,
        CompletenessLevel,
        AccuracyLevel,
    )
    __all__.extend([
        "DataAssetMetadata",
        "AIModelMetadata",
        "BusinessEntityMetadata",
        "WorkflowMetadata",
        "DataLineageRelation",
        "QualityMetrics",
        "MetadataSearchResult",
        "MetadataFilter",
        "MetadataResponse",
        "TechnicalMetadata",
        "BusinessMetadata",
        "OperationalMetadata",
        "DataAssetType",
        "ModelType",
        "EntityType",
        "WorkflowType",
        "LineageRelationType",
        "QualityLevel",
        "QualityScore",
        "CompletenessLevel",
        "AccuracyLevel",
    ])
except (ImportError, AttributeError) as e:
    # 如果模型不存在，不导出，但不影响包的导入
    pass
EOF

# 验证文件已创建
echo "File created. First 10 lines:"
head -10 shared_libs/__init__.py

# 重启服务
echo ""
echo "Restarting mcp-gateway..."
sudo docker compose restart mcp-gateway

# 等待启动
echo "Waiting for service to start..."
sleep 20

# 检查状态
echo ""
echo "=== Service Status ==="
sudo docker compose ps mcp-gateway

# 查看日志
echo ""
echo "=== Recent Logs ==="
sudo docker compose logs --tail=30 mcp-gateway

# 测试服务
echo ""
echo "=== Testing Service ==="
curl -s http://localhost:8001/api/health
