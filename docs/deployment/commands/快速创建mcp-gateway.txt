# 快速创建并启动mcp-gateway
# 复制执行

cd /opt/enterprise-ai-platform

# 一键启动所有服务
sudo docker start enterprise-ai-postgres && sleep 15 && sudo docker compose up -d redis mcp-gateway && sleep 20 && sudo docker ps

# 查看mcp-gateway日志
sudo docker compose logs --tail=30 mcp-gateway

# 测试服务
curl http://localhost:8001/api/health

# 如果健康检查失败，查看详细日志
sudo docker compose logs mcp-gateway
