# 快速启动 - 避免postgres被停止
# 复制执行以下命令

cd /opt/enterprise-ai-platform

# 方法1: 使用docker start启动postgres（最简单）
sudo docker start enterprise-ai-postgres && sleep 15 && sudo docker compose up -d redis mcp-gateway && sleep 15 && sudo docker compose ps

# 方法2: 使用docker compose一次性启动（推荐）
sudo docker compose up -d postgres redis mcp-gateway && sleep 20 && sudo docker compose ps

# 查看mcp-gateway日志
sudo docker compose logs --tail=30 mcp-gateway

# 测试服务
curl http://localhost:8001/api/health
