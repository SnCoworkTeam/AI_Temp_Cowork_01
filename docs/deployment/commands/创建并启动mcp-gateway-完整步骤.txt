# 创建并启动mcp-gateway的完整步骤
# mcp-gateway容器不存在，需要创建

cd /opt/enterprise-ai-platform

# ============================================
# 步骤1: 启动postgres（使用docker start）
# ============================================
sudo docker start enterprise-ai-postgres

# 等待postgres就绪（约15秒）
sleep 15

# 验证postgres运行
sudo docker ps | grep postgres

# 测试postgres连接
sudo docker exec enterprise-ai-postgres pg_isready -U postgres

# ============================================
# 步骤2: 启动redis服务
# ============================================
# 检查redis容器是否存在
sudo docker ps -a | grep enterprise-ai-redis

# 启动redis（如果不存在会自动创建）
sudo docker compose up -d redis

# 等待redis就绪（约5秒）
sleep 5

# 验证redis运行
sudo docker ps | grep enterprise-ai-redis

# 测试redis连接
sudo docker exec enterprise-ai-redis redis-cli ping

# ============================================
# 步骤3: 创建并启动mcp-gateway
# ============================================
# 检查mcp-gateway镜像是否存在
sudo docker images | grep mcp-gateway

# 如果镜像不存在，先构建
# sudo docker compose build mcp-gateway

# 创建并启动mcp-gateway
sudo docker compose up -d mcp-gateway

# 等待mcp-gateway启动（约15-30秒，首次启动可能需要更长时间）
sleep 20

# ============================================
# 步骤4: 检查所有服务状态
# ============================================
sudo docker ps

# 使用docker compose查看
sudo docker compose ps

# ============================================
# 步骤5: 查看mcp-gateway日志
# ============================================
sudo docker compose logs --tail=50 mcp-gateway

# 如果启动失败，查看完整日志
sudo docker compose logs mcp-gateway

# ============================================
# 步骤6: 测试服务
# ============================================
curl http://localhost:8001/api/health

# 如果健康检查失败，等待更长时间后重试
sleep 10
curl http://localhost:8001/api/health

# ============================================
# 一键启动命令
# ============================================
sudo docker start enterprise-ai-postgres && sleep 15 && sudo docker compose up -d redis && sleep 5 && sudo docker compose up -d mcp-gateway && sleep 20 && sudo docker ps && echo "" && echo "Testing mcp-gateway..." && curl http://localhost:8001/api/health
