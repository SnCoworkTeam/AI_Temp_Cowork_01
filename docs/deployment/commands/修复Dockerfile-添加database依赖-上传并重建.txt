# 修复Dockerfile.dev - 添加database依赖
# 问题：Dockerfile.dev没有安装database/requirements.txt，导致缺少sqlalchemy等依赖
# 解决：在Dockerfile.dev中添加安装database依赖的步骤

# ============================================
# 第一步：在本地PowerShell上传修复后的Dockerfile.dev
# ============================================
# 在本地Windows PowerShell中执行：

scp -i "E:\enterprise-ai-platform\enterprise_ai_platform.pem" -o StrictHostKeyChecking=no mcp-gateway/Dockerfile.dev ubuntu@43.143.139.197:/opt/enterprise-ai-platform/mcp-gateway/

# ============================================
# 第二步：在服务器上重新构建并启动容器
# ============================================
# SSH到服务器后执行：

cd /opt/enterprise-ai-platform

# 1. 停止并删除mcp-gateway容器
sudo docker compose stop mcp-gateway
sudo docker compose rm -f mcp-gateway

# 2. 重新构建并启动（使用新的Dockerfile.dev）
sudo docker compose up -d --build mcp-gateway

# 3. 等待构建和启动（可能需要几分钟）
echo "Building and starting... This may take a few minutes..."
sleep 60

# 4. 检查状态
sudo docker compose ps mcp-gateway

# 5. 查看日志（应该没有sqlalchemy错误了）
sudo docker compose logs --tail=30 mcp-gateway

# 6. 测试服务
curl http://localhost:8001/api/health

# 一键执行命令（服务器上）
