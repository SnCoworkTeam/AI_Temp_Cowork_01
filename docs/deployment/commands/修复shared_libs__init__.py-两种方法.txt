# 修复shared_libs/__init__.py - 两种方法

# ============================================
# 方法1：在本地PowerShell上传（推荐）
# ============================================
# 在本地Windows PowerShell中执行：

# 方式A：使用脚本
powershell -ExecutionPolicy Bypass -File .\上传修复__init__.py.ps1

# 方式B：手动执行
scp -i "E:\enterprise-ai-platform\enterprise_ai_platform.pem" -o StrictHostKeyChecking=no shared_libs/__init__.py ubuntu@43.143.139.197:/opt/enterprise-ai-platform/shared_libs/

# ============================================
# 方法2：直接在服务器上创建修复后的文件（如果上传不方便）
# ============================================
# 在服务器上执行以下命令创建修复后的文件：

cd /opt/enterprise-ai-platform

# 备份原文件
cp shared_libs/__init__.py shared_libs/__init__.py.bak

# 创建修复后的文件
cat > shared_libs/__init__.py << 'EOF'
"""共享库包"""
# 确保包可以被导入，即使子模块导入失败
__all__ = []

# 导出元数据模型（如果存在）
try:
    from .src.models import (
        DataAssetMetadata,
        AIModelMetadata,
        BusinessEntityMetadata,
        WorkflowMetadata,
        DataLineageRelation,
        QualityMetrics,
        MetadataSearchResult,
        MetadataFilter,
        MetadataResponse,
        TechnicalMetadata,
        BusinessMetadata,
        OperationalMetadata,
        DataAssetType,
        ModelType,
        EntityType,
        WorkflowType,
        LineageRelationType,
        QualityLevel,
        QualityScore,
        CompletenessLevel,
        AccuracyLevel,
    )
    __all__.extend([
        "DataAssetMetadata",
        "AIModelMetadata",
        "BusinessEntityMetadata",
        "WorkflowMetadata",
        "DataLineageRelation",
        "QualityMetrics",
        "MetadataSearchResult",
        "MetadataFilter",
        "MetadataResponse",
        "TechnicalMetadata",
        "BusinessMetadata",
        "OperationalMetadata",
        "DataAssetType",
        "ModelType",
        "EntityType",
        "WorkflowType",
        "LineageRelationType",
        "QualityLevel",
        "QualityScore",
        "CompletenessLevel",
        "AccuracyLevel",
    ])
except (ImportError, AttributeError) as e:
    # 如果模型不存在，不导出，但不影响包的导入
    pass
EOF

# 验证文件
cat shared_libs/__init__.py | head -10

# 重启服务
sudo docker compose restart mcp-gateway

# 等待启动
sleep 20

# 检查日志
sudo docker compose logs --tail=30 mcp-gateway

# ============================================
# 一键执行（方法2，在服务器上）
# ============================================
cd /opt/enterprise-ai-platform && cp shared_libs/__init__.py shared_libs/__init__.py.bak && cat > shared_libs/__init__.py << 'ENDOFFILE'
"""共享库包"""
__all__ = []
try:
    from .src.models import (
        DataAssetMetadata, AIModelMetadata, BusinessEntityMetadata, WorkflowMetadata,
        DataLineageRelation, QualityMetrics, MetadataSearchResult, MetadataFilter,
        MetadataResponse, TechnicalMetadata, BusinessMetadata, OperationalMetadata,
        DataAssetType, ModelType, EntityType, WorkflowType, LineageRelationType,
        QualityLevel, QualityScore, CompletenessLevel, AccuracyLevel,
    )
    __all__.extend([
        "DataAssetMetadata", "AIModelMetadata", "BusinessEntityMetadata", "WorkflowMetadata",
        "DataLineageRelation", "QualityMetrics", "MetadataSearchResult", "MetadataFilter",
        "MetadataResponse", "TechnicalMetadata", "BusinessMetadata", "OperationalMetadata",
        "DataAssetType", "ModelType", "EntityType", "WorkflowType", "LineageRelationType",
        "QualityLevel", "QualityScore", "CompletenessLevel", "AccuracyLevel",
    ])
except (ImportError, AttributeError):
    pass
ENDOFFILE
&& sudo docker compose restart mcp-gateway && sleep 20 && sudo docker compose logs --tail=20 mcp-gateway
