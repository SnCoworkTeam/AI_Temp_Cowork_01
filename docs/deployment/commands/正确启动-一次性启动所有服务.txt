# 正确启动方法 - 一次性启动所有服务
# 问题：单独启动mcp-gateway会导致postgres被停止
# 解决：使用docker compose一次性启动所有相关服务

cd /opt/enterprise-ai-platform

# ============================================
# 方法1: 一次性启动所有服务（推荐）
# ============================================
# Docker Compose会按依赖顺序启动，不会停止已运行的服务
sudo docker compose up -d postgres redis mcp-gateway

# 等待所有服务启动（约20秒）
sleep 20

# 检查所有服务状态
sudo docker compose ps

# 查看mcp-gateway日志
sudo docker compose logs --tail=50 mcp-gateway

# 测试服务
curl http://localhost:8001/api/health

# ============================================
# 方法2: 分步启动（如果方法1失败）
# ============================================
# 步骤1: 启动postgres
sudo docker compose up -d postgres

# 等待postgres就绪
sleep 15

# 步骤2: 启动redis
sudo docker compose up -d redis

# 等待redis就绪
sleep 5

# 步骤3: 启动mcp-gateway（此时postgres和redis已在docker compose管理下）
sudo docker compose up -d mcp-gateway

# 等待mcp-gateway启动
sleep 15

# 检查状态
sudo docker compose ps

# ============================================
# 方法3: 如果mcp-gateway已经启动，只需重启postgres
# ============================================
# 检查mcp-gateway是否已经运行
sudo docker ps | grep mcp-gateway

# 如果mcp-gateway已运行，只需启动postgres
sudo docker compose up -d postgres

# 等待postgres就绪
sleep 15

# 验证所有服务
sudo docker compose ps

# ============================================
# 一键启动命令（推荐）
# ============================================
sudo docker compose up -d postgres redis mcp-gateway && sleep 20 && sudo docker compose ps && echo "" && echo "Testing mcp-gateway..." && curl http://localhost:8001/api/health
