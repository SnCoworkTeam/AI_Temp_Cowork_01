# 立即修复 - 上传docker-compose.yml并重新创建容器
# 在本地PowerShell执行上传，然后在服务器执行重建

# ============================================
# 第一步：在本地PowerShell上传docker-compose.yml
# ============================================
scp -i "E:\enterprise-ai-platform\enterprise_ai_platform.pem" -o StrictHostKeyChecking=no docker-compose.yml ubuntu@43.143.139.197:/opt/enterprise-ai-platform/

# ============================================
# 第二步：在服务器上执行（SSH后执行）
# ============================================
cd /opt/enterprise-ai-platform

# 1. 确认.env文件已修复
grep LOG_LEVEL .env

# 2. 停止并删除mcp-gateway容器（必须删除，不能只重启）
sudo docker compose stop mcp-gateway
sudo docker compose rm -f mcp-gateway

# 3. 重新创建并启动mcp-gateway（使用新的docker-compose.yml配置）
sudo docker compose up -d --build mcp-gateway

# 4. 等待启动
sleep 20

# 5. 检查状态
sudo docker compose ps mcp-gateway

# 6. 查看日志（应该没有错误了）
sudo docker compose logs --tail=30 mcp-gateway

# 7. 测试服务
curl http://localhost:8001/api/health

# ============================================
# 一键执行命令（服务器上）
# ============================================
cd /opt/enterprise-ai-platform && grep LOG_LEVEL .env && sudo docker compose stop mcp-gateway && sudo docker compose rm -f mcp-gateway && sudo docker compose up -d --build mcp-gateway && sleep 20 && sudo docker compose ps mcp-gateway && echo "" && sudo docker compose logs --tail=20 mcp-gateway
