```python
# database/src/models/enterprise_architecture_models.py

class OrganizationUnit(BaseModel, TimestampMixin):
    """组织单元模型"""
    __tablename__ = "organization_units"
    
    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4, comment="组织ID")
    name = Column(String(255), nullable=False, index=True, comment="组织名称")
    code = Column(String(100), nullable=True, unique=True, index=True, comment="组织编码")
    description = Column(Text, nullable=True, comment="组织描述")
    organization_type = Column(String(100), nullable=True, comment="组织类型")
    level = Column(Integer, nullable=True, default=1, comment="组织层级")
    parent_id = Column(UUID(as_uuid=True), ForeignKey("organization_units.id", ondelete="SET NULL"), nullable=True, index=True, comment="上级组织ID")
    manager_id = Column(UUID(as_uuid=True), ForeignKey("users.id"), nullable=True, index=True, comment="负责人ID")
    location = Column(String(255), nullable=True, comment="地理位置")
    status = Column(String(50), nullable=True, default="active", index=True, comment="状态")
    neo4j_node_id = Column(String(255), nullable=True, comment="Neo4j节点ID")
    meta_data = Column(JSONB, nullable=True, default=dict, comment="扩展元数据")
    
    # 关系
    parent = relationship("OrganizationUnit", remote_side=[id], backref="children")
    
    __table_args__ = (
        Index('idx_organization_units_name', 'name'),
        Index('idx_organization_units_code', 'code'),
        Index('idx_organization_units_status', 'status'),
    )
    
    def __repr__(self):
        return f"<OrganizationUnit(id={self.id}, name={self.name})>"
```

## 🗺️ 三、Neo4j图模型设计

### 3.1 组织节点创建

```python
# metadata-service/src/services/organization_architecture_graph_service.py

class OrganizationArchitectureGraphService:
    """组织架构图服务"""
    
    def __init__(self, db: Session, neo4j_client: Neo4jClient):
        self.db = db
        self.neo4j = neo4j_client
    
    async def create_organization_unit_node(
        self, org_unit: OrganizationUnit
    ) -> str:
        """创建组织单元节点"""
        node_id = await self.neo4j.create_node(
            labels=["OrganizationUnit"],
            properties={
                "uuid": str(org_unit.id),
                "name": org_unit.name,
                "code": org_unit.code,
                "description": org_unit.description,
                "organization_type": org_unit.organization_type,
                "level": org_unit.level,
                "location": org_unit.location,
                "status": org_unit.status
            }
        )
        
        org_unit.neo4j_node_id = node_id
        self.db.commit()
        
        return node_id
    
    async def create_organization_hierarchy(
        self, org_unit: OrganizationUnit
    ):
        """创建组织层级关系"""
        if org_unit.parent_id:
            parent = self.db.query(OrganizationUnit).filter(
                OrganizationUnit.id == org_unit.parent_id
            ).first()
            
            if parent and parent.neo4j_node_id and org_unit.neo4j_node_id:
                await self.neo4j.create_relationship(
                    source_id=parent.neo4j_node_id,
                    target_id=org_unit.neo4j_node_id,
                    rel_type="PARENT_OF",
                    properties={
                        "created_at": datetime.utcnow().isoformat()
                    }
                )
```

```python
async def link_organization_to_business_capability(
```

## 📊 四、查询场景设计

### 4.1 组织责任查询

```cypher
// 查询：采购部负责哪些业务能力？
MATCH (org:OrganizationUnit {name: "采购部"})-[:OWNS]->(cap:BusinessCapability)
RETURN cap.name as capability, cap.description as description

// 查询：采购部执行哪些业务流程？
MATCH (org:OrganizationUnit {name: "采购部"})-[:EXECUTES]->(proc:BusinessProcess)
RETURN proc.name as process, proc.criticality as criticality

// 查询：采购部使用哪些系统？
MATCH (org:OrganizationUnit {name: "采购部"})-[:USES]->(sys:ApplicationSystem)
RETURN sys.name as system, sys.system_type as type
```

### 4.2 组织资源统计

```cypher
// 查询：每个部门使用的系统数量
MATCH (org:OrganizationUnit)-[:USES]->(sys:ApplicationSystem)
RETURN org.name as department, count(sys) as system_count
ORDER BY system_count DESC

// 查询：每个部门的IT投资（基于系统数量）
MATCH (org:OrganizationUnit)-[:USES]->(sys:ApplicationSystem)
WITH org, count(sys) as system_count
RETURN org.name as department, system_count, 
       system_count * 100000 as estimated_investment
```

### 4.3 组织变革影响分析

```cypher
// 查询：如果采购部和供应链部合并，会使用哪些系统？
MATCH (org1:OrganizationUnit {name: "采购部"})-[:USES]->(sys:ApplicationSystem)
MATCH (org2:OrganizationUnit {name: "供应链部"})-[:USES]->(sys)
RETURN sys.name as shared_system, 
       count(DISTINCT sys) as overlap_count

// 查询：部门合并后需要整合的系统
MATCH (org1:OrganizationUnit {name: "采购部"})-[:USES]->(sys1:ApplicationSystem)
MATCH (org2:OrganizationUnit {name: "供应链部"})-[:USES]->(sys2:ApplicationSystem)
WHERE sys1 <> sys2
RETURN collect(DISTINCT sys1.name) + collect(DISTINCT sys2.name) as systems_to_integrate
```
