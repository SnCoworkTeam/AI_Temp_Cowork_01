**1. 缓存策略**
```python
# project-management/src/services/critical_path_calculator.py
from functools import lru_cache
from datetime import datetime, timedelta
import hashlib
import json

class CriticalPathCalculator:
    """关键路径计算器（带缓存）"""
    
    CACHE_TTL = timedelta(minutes=5)  # 缓存5分钟
    
    @staticmethod
    def _get_cache_key(plan_id: UUID, task_ids: List[UUID]) -> str:
        """生成缓存键"""
        task_ids_str = ','.join(sorted([str(tid) for tid in task_ids]))
        return f"critical_path:{plan_id}:{hashlib.md5(task_ids_str.encode()).hexdigest()}"
    
    @classmethod
    def calculate_critical_path_cached(cls, plan_id: UUID, db: Session, redis_client=None):
        """带缓存的关键路径计算"""
        # 如果使用Redis
        if redis_client:
            cache_key = cls._get_cache_key(plan_id, [])
            cached_result = redis_client.get(cache_key)
            if cached_result:
                return json.loads(cached_result)
        
        # 计算关键路径
        result = cls.calculate_critical_path(plan_id, db)
        
        # 缓存结果
        if redis_client:
            redis_client.setex(
                cache_key,
                int(cls.CACHE_TTL.total_seconds()),
                json.dumps(result)
            )
        
        return result
    
    @classmethod
    def invalidate_cache(cls, plan_id: UUID, redis_client=None):
        """使缓存失效"""
        if redis_client:
            pattern = f"critical_path:{plan_id}:*"
            for key in redis_client.scan_iter(match=pattern):
                redis_client.delete(key)
```

    """冲突解决器"""
    
    @staticmethod
    def detect_conflicts(plan_task: ProjectPlanTask, actual_task: Task) -> List[dict]:
        """检测冲突"""
        conflicts = []
        
        if plan_task.name != actual_task.name:
            conflicts.append({
                "field": "name",
                "plan_value": plan_task.name,
                "actual_value": actual_task.name,
                "type": "mismatch"
            })
        
        if plan_task.end_date and actual_task.due_date:
            if plan_task.end_date != actual_task.due_date:
                conflicts.append({
                    "field": "due_date",
                    "plan_value": str(plan_task.end_date),
                    "actual_value": str(actual_task.due_date),
                    "type": "date_mismatch"

        elif resolution == "use_actual":
            # 使用实际值
            setattr(plan_task, field, getattr(actual_task, field))

from fastapi import HTTPException, Depends

def require_plan_permission(permission: PlanPermission):
    """要求计划权限的装饰器"""
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            # 获取当前用户和计划ID
            current_user = kwargs.get('current_user')
            plan_id = kwargs.get('plan_id')

            return await func(*args, **kwargs)
        return wrapper
